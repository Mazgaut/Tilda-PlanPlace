async function generate_pdf(imgs, type) {
    if (typeof custom_pdf_generation === 'function') {
        return custom_pdf_generation(type)
    }

    async function ask_pdf_photo_mode() {
        return new Promise(resolve => {
            if (!pp.interface?.root?.$refs?.pdf_photo_mode_modal) {
                resolve('auto');
                return;
            }
            pp.interface.root.$refs.pdf_photo_mode_modal.show_modal(resolve);
        });
    }

    async function ask_make_photo(photos_left) {
        return new Promise(resolve => {
            if (!pp.interface?.root?.$refs?.make_photo_modal) {
                resolve('go_to_spec');
                return;
            }
            pp.interface.root.$refs.make_photo_modal.show_modal(resolve, photos_left);
        });
    }

    let photos = [];
    if (constructor_settings.own_photo_pdf && !imgs) {
        let photo_mode = await ask_pdf_photo_mode();
        if (photo_mode === 'custom_photos') {
            for (let i = 5; i > 0; i--) {
                let make_photo = await ask_make_photo(i);
                if (make_photo === 'make_photo') {
                    photos.push(get_image());
                } else if (make_photo === 'go_to_spec') {
                    break;
                } else {
                    return;
                }
            }
        } else if (photo_mode !== 'auto') {
            return;
        }
    }

    await check_pdf_lib()

    if (!md || typeof md !== 'object') {
        md = {sum: {}, all: {module: {}}};
    } else {
        if (md.sum == null || typeof md.sum !== 'object') {
            md.sum = {};
        }
        if (md.all == null || typeof md.all !== 'object') {
            md.all = {module: {}};
        } else if (md.all.module == null || typeof md.all.module !== 'object') {
            md.all.module = {};
        }

    let p_width = global_options.pdf.width;
    let p_height = global_options.pdf.height;

    let img_size_offset_x = global_options.pdf.img_size_offset_x
    let img_size_offset_y = global_options.pdf.img_size_offset_y

        },
        vLineColor: function (i, node) {
            return 'gray';

        paddingLeft: function (i, node) {

        },

                            obj.width + ' м',
                            round_decimals_2(obj.total_count) + ' ' + lang['units_' + obj.price_type],
                        )

                        if (show_price2) row.push({
                            text: format_price(obj.total_price),
                            alignment: 'right'
                        });
                        tabledata.push(row)
                    }

                    dd.content.push(
                        {
                            table: {
                                headerRows: 1,
                                widths: widths,
                                // dontBreakRows: true,
                                // keepWithHeaderRows: 1,
                                body: tabledata,
                            },
                            layout: tables_layout
                        },
                        {
                            text: '\n\n'
                        }
                    );
                }

                if (key == 'item') {
                    let header_row = [
                        {
                            text: lang['title'],
                            style: 'tableHeader'
                        },
                        {
                            text: lang['code'],
                            style: 'tableHeader'
                        },
                        {
                            text: lang['count_short'],
                            style: 'tableHeader'
                        },
                    ]
                    if (show_price2) header_row.push({
                        text: lang['price'],
                        style: 'tableHeader'
                    });

                    tabledata.push(header_row);
                    dd.content.push({
                        text: lang['accessories_furniture'] + ':' + '\n\n',
                        style: 'tableHeader'
                    });

                    for (const [i_key, obj] of Object.entries(item)) {
                        console.log(i_key, obj);
                        let row = []
                        const pdf_safe_obj_name = ('' + (obj.name || ''))
                            .replace(/\u00A0/g, ' ')
                            .replace(/\u25CF/g, '\u2022');
                        row.push(
                            obj.is_compound_header ? {text: pdf_safe_obj_name, bold: true} : pdf_safe_obj_name,
                            obj.code,
                            round_decimals_2(obj.total_count) + ' ' + lang['units_' + obj.price_type],
                        )

                        if (show_price2) row.push({
                            text: (obj.is_compound_header && (parseFloat(obj.total_price) || 0) <= 0)
                                ? (lang['compound'] || 'Составная')
                                : format_price(obj.total_price),
                            alignment: 'right'
                        });
                        tabledata.push(row)
                    }

                    let widths_item = [];
                    for (let ii = 0; ii < header_row.length; ii++) {
                        ii == 0 ? widths_item.push('*') : widths_item.push('auto')
                    }

                    dd.content.push(
                        {
                            table: {
                                headerRows: 1,
                                widths: widths_item,
                                // dontBreakRows: true,
                                // keepWithHeaderRows: 1,
                                body: tabledata,
                            },
                            layout: tables_layout
                        },
                        {
                            text: '\n\n'
                        }
                    );
                }
            }

            if (pdf_settings.report_modules_comments_list !== false) {
                if (obj.comments && String(obj.comments).trim() !== '') {
                    dd.content.push({
                        text: lang['module_comments'] + ':',
                        style: 'tableHeader',
                    });

                    dd.content.push(
                        {
                            text: obj.comments,
                            alignment: 'left'
                        },
                        {
                            text: '\n\n'
                        }
                    );
                }
            }

            if (show_price2) dd.content.push(
                {
                    text: lang['price_module'] + ': ' + format_price(obj.price) + '\n\n',
                    style: 'tableHeader',
                    bold: false,
                    italics: true,
                    alignment: 'left'
                },
                {
                    text: '\n',
                }
            );

        }
    }

    if (typeof pdf_generation_callback === 'function') {
        pdf_generation_callback(dd)
    }
    if (!imgs) images = null;


    console.log(dd)

    const pdfDocGenerator = await pdfMake.createPdf(dd);

    switch (type) {
        case 'all':
            return {
                base64: new Promise(function (resolve, reject) {
                    pdfDocGenerator.getDataUrl(function (dataUrl) {
                        resolve(dataUrl);
                    });
                }),
                blob: new Promise(function (resolve, reject) {
                    pdfDocGenerator.getBlob(function (blob) {
                        resolve(blob);
                    });
                })
            }
        case 'blob':
            return new Promise(function (resolve, reject) {
                pdfDocGenerator.getBlob(function (blob) {
                    resolve(blob);
                });
            });
        default:
            return new Promise(function (resolve, reject) {
                pdfDocGenerator.getDataUrl(function (dataUrl) {
                    resolve(dataUrl);
                });
            });
    }


    if (action === 'send') {
        return new Promise(function (resolve, reject) {
            pdfDocGenerator.getDataUrl(function (dataUrl) {
                resolve(dataUrl);
            });
        });
    } else if (action === 'blob') {
        return new Promise(function (resolve, reject) {
            pdfDocGenerator.getBlob(function (blob) {
                resolve(blob);
            });
        });
    } else if (action === 'new_tab') {
        pdfDocGenerator.getDataUrl((dataUrl) => {
            open_data_uri_window(dataUrl)
        });
    } else if (action === 'download') {
        let date = new Date();
        let str = '' + date.getDate() + '.' + (date.getMonth() + 1) + '.' + date.getFullYear() + '_' + date.getHours() + ':' + date.getMinutes() + '.pdf';

        pdfDocGenerator.download(str);
    }
}
